function doGet(e) {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  return ContentService.createTextOutput(JSON.stringify({
    jadwal: getSheetData(ss, "Jadwal"),
    kunjungan: getSheetData(ss, "Kunjungan"),
    saran: getSheetData(ss, "Saran")
  })).setMimeType(ContentService.MimeType.JSON);
}

function doPost(e) {
  try {
    var data = JSON.parse(e.postData.contents);
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    var action = data.action;
    
    if (action === "addJadwal") appendRow(ss, "Jadwal", [data.id, data.hari, data.mulai, data.selesai, data.kelas, data.mapel, data.guru]);
    else if (action === "deleteJadwal") deleteRowById(ss, "Jadwal", data.id);
    else if (action === "addKunjungan") appendRow(ss, "Kunjungan", [data.id, data.tanggal, data.jam, data.guru, data.kelas, data.jmlSiswa, data.mapel, data.pcUsed, data.kondisi, data.catatan]);
    else if (action === "deleteKunjungan") deleteRowById(ss, "Kunjungan", data.id);
    else if (action === "addSaran") appendRow(ss, "Saran", [data.id, data.nama, data.peran, data.kategori, data.pesan, data.tanggal, data.tanggapan || '', data.status || 'Proses']);
    else if (action === "updateSaran") updateSaranRow(ss, data.id, data.tanggapan, data.status);
    else if (action === "deleteSaran") deleteRowById(ss, "Saran", data.id);
    
    return ContentService.createTextOutput(JSON.stringify({ status: "success" })).setMimeType(ContentService.MimeType.JSON);
  } catch (err) {
    return ContentService.createTextOutput(JSON.stringify({ status: "error", message: err.toString() })).setMimeType(ContentService.MimeType.JSON);
  }
}

function getSheetData(ss, sheetName) {
  var sheet = ss.getSheetByName(sheetName);
  if (!sheet) return [];
  var rows = sheet.getDataRange().getValues();
  if (rows.length < 2) return [];
  var headers = rows[0];
  var result = [];
  for (var i = 1; i < rows.length; i++) {
    var obj = {};
    for (var j = 0; j < headers.length; j++) {
      obj[headers[j]] = rows[i][j];
    }
    result.push(obj);
  }
  return result;
}

function appendRow(ss, sheetName, rowData) {
  var sheet = ss.getSheetByName(sheetName);
  if (!sheet) {
    sheet = ss.insertSheet(sheetName);
    var headers = sheetName === "Jadwal" ? ["id","hari","mulai","selesai","kelas","mapel","guru"] :
                  sheetName === "Kunjungan" ? ["id","tanggal","jam","guru","kelas","jmlSiswa","mapel","pcUsed","kondisi","catatan"] :
                  ["id","nama","peran","kategori","pesan","tanggal","tanggapan","status"];
    sheet.appendRow(headers);
  }
  sheet.appendRow(rowData);
}

function deleteRowById(ss, sheetName, id) {
  var sheet = ss.getSheetByName(sheetName);
  if (!sheet) return;
  var rows = sheet.getDataRange().getValues();
  for (var i = 1; i < rows.length; i++) {
    if (String(rows[i][0]) === String(id)) {
      sheet.deleteRow(i + 1);
      break;
    }
  }
}

function updateSaranRow(ss, id, tanggapan, status) {
  var sheet = ss.getSheetByName("Saran");
  if (!sheet) return;
  var rows = sheet.getDataRange().getValues();
  for (var i = 1; i < rows.length; i++) {
    if (String(rows[i][0]) === String(id)) {
      sheet.getRange(i + 1, 7).setValue(tanggapan);
      sheet.getRange(i + 1, 8).setValue(status);
      break;
    }
  }
}
